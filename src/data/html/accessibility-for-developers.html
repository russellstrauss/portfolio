Focus Management

// Trap focus in modal function trapFocus(container) { const focusableElements = container.querySelectorAll( 'a[href], button:not([disabled]), textarea, input:not([disabled]), select, [tabindex]:not([tabindex="-1"])' ); const firstElement = focusableElements[0]; const lastElement = focusableElements[focusableElements.length - 1]; container.addEventListener('keydown', (e) => { if (e.key === 'Tab') { if (e.shiftKey) { if (document.activeElement === firstElement) { e.preventDefault(); lastElement.focus(); } } else { if (document.activeElement === lastElement) { e.preventDefault(); firstElement.focus(); } } } }); } // Return focus to trigger element function openModal(trigger, modal) { trigger.setAttribute('data-previous-focus', document.activeElement.id); modal.show(); const firstFocusable = modal.querySelector('[tabindex="0"], button, a, input'); firstFocusable?.focus(); } function closeModal(modal) { const triggerId = modal.getAttribute('data-previous-focus'); const trigger = document.getElementById(triggerId); modal.hide(); trigger?.focus(); }

Dynamic Content Updates

ARIA Live Regions

// Announce content changes to screen readers function announceToScreenReader(message, priority = 'polite') { const liveRegion = document.getElementById('live-region') || createLiveRegion(priority); liveRegion.textContent = message; setTimeout(() => { // Clear after announcement liveRegion.textContent = ''; }, 1000); } function createLiveRegion(priority) { const region = document.createElement('div'); region.id = 'live-region'; region.setAttribute('role', 'status'); region.setAttribute('aria-live', priority); region.setAttribute('aria-atomic', 'true'); region.className = 'sr-only'; document.body.appendChild(region); return region; }

Accessible Error Handling

function validateForm(form) { const errors = []; const inputs = form.querySelectorAll('input, textarea, select'); inputs.forEach(input => { const errorId = input.id + '-error'; let errorElement = document.getElementById(errorId); if (!input.validity.valid) { if (!errorElement) { errorElement = document.createElement('span'); errorElement.id = errorId; errorElement.className = 'error'; errorElement.setAttribute('role', 'alert'); errorElement.setAttribute('aria-live', 'polite'); input.parentNode.appendChild(errorElement); } input.setAttribute('aria-invalid', 'true'); input.setAttribute('aria-describedby', errorId); errorElement.textContent = getErrorMessage(input); errors.push(input); } else { input.removeAttribute('aria-invalid'); input.removeAttribute('aria-describedby'); if (errorElement) { errorElement.remove(); } } }); if (errors.length > 0) { errors[0].focus(); announceToScreenReader('Form has errors. Please review and correct.'); } return errors.length === 0; }

Testing in Development

Quick Accessibility Checks

  • Tab through the entire page - can you access everything?
  • Check heading structure with a browser extension
  • Validate HTML markup
  • Test with browser zoom at 200%
  • Check color contrast with browser DevTools
  • Review ARIA usage with accessibility tree viewer

Development Tools

  • Chrome Accessibility Tree (DevTools > Elements > Accessibility)
  • Firefox Accessibility Inspector
  • axe DevTools browser extension
  • WAVE browser extension
  • Lighthouse accessibility audit

Common Pitfalls to Avoid

  • Using div/span for interactive elements: Use button, a, or proper ARIA roles
  • Removing focus outlines: Style them instead of removing
  • Relying on color alone: Add icons, text, or patterns
  • Poor heading hierarchy: Don't skip levels (h1 to h3)
  • Missing alt text: All images need appropriate alt attributes
  • Keyboard traps: Ensure users can always navigate away
  • Auto-playing media: Provide controls and don't auto-play audio
  • Low contrast: Ensure text meets 4.5:1 contrast ratio

Integration with Frameworks

React

  • Use semantic HTML elements
  • Manage focus with useRef and useEffect
  • Use proper ARIA attributes
  • Test with @testing-library/react (includes accessibility queries)
  • Consider react-aria or Reach UI for accessible components

Vue

  • Use semantic HTML in templates
  • Manage focus with $refs and lifecycle hooks
  • Use v-bind for dynamic ARIA attributes
  • Test with @vue/test-utils and accessibility assertions

Continuous Improvement

  • Include accessibility in code review process
  • Run automated tests in CI/CD
  • Keep up with WCAG updates
  • Learn from accessibility audits
  • Share knowledge with team members
